% ======================================================
%  ECE 457A FINAL Cheat Sheet Layout Definition
%  (same style as the midterm crib sheet)
% ======================================================

% --- Required packages ---
\usepackage{xcolor}
\usepackage{fancyhdr}
\usepackage{titlesec}

% --- Colors and styles ---
\definecolor{myblue}{cmyk}{1,.72,0,.38}
\definecolor{myred}{cmyk}{0,.9,.8,.06}
\colorlet{circle edge}{myblue}
\colorlet{circle area}{myblue!5}

\tcbset{
  cheatbox/.style={
    enhanced,
    colback=white,
    fontupper=\sffamily,
    colframe=gray,
    boxrule=0.5pt,
    arc=2pt,
    left=4pt,right=4pt,top=2pt,bottom=2pt,
  }
}

% --- Make math blue everywhere ---
\everymath\expandafter{\the\everymath \color{myblue}}
\everydisplay\expandafter{\the\everydisplay \color{myblue}}

% --- Compact layout ---
\renewcommand{\baselinestretch}{.9}
\setlength{\parindent}{0pt}
\setlength{\columnsep}{8pt}
\setlength{\emergencystretch}{3em}
\hbadness=10000
\AtBeginDocument{\raggedright}

% --- Header / Footer setup ---
\pagestyle{fancy}
\fancyhf{}
\fancyhead[C]{\footnotesize\sffamily ECE 457A --- Final Cheatsheet \quad--\quad Page \thepage}
\renewcommand{\headrulewidth}{0.4pt}
\fancyfoot{}

% --- Section formatting (sized to match the \scriptsize body: headers were
%     oversized because they stayed at \small/\footnotesize on a smaller body) ---
% Headers sized to the (tiny) body so nothing looks oversized: section is
% distinguished by blue+bold, subsection by bold, both at body \scriptsize.
\titleformat*{\section}{\color{myblue}\sffamily\scriptsize\bfseries}
\titleformat*{\subsection}{\sffamily\bfseries\scriptsize}
\titleformat*{\subsubsection}{\sffamily\itshape}

% --- Tight spacing before/after sections (no new line after heading feel) ---
\titlespacing*{\section}{0pt}{.4ex}{.3ex}
\titlespacing*{\subsection}{0pt}{.3ex}{.2ex}

% --- Run-in headings: title and body share one line (no line break after) ---
\newcommand{\runsec}[1]{\par\addvspace{.5ex}{\color{myblue}\sffamily\scriptsize\bfseries #1}\ \ignorespaces}
\newcommand{\runsub}[1]{\par\addvspace{.3ex}{\sffamily\scriptsize\bfseries #1}\ \ignorespaces}

% --- Helper: small section header box ---
\newcommand{\headerbox}[1]{%
  \begin{mdframed}[linecolor=gray,linewidth=0.5pt,
                   leftmargin=0mm,rightmargin=0mm,
                   skipbelow=0mm,skipabove=1mm]
  \footnotesize\sffamily #1
  \end{mdframed}
}

% --- Helper: fit a tikz picture to the column width ---
\newcommand{\fitfig}[1]{\begin{center}\resizebox{0.98\linewidth}{!}{#1}\end{center}}

% --- Common tikz node styles for diagrams ---
\tikzset{
  snode/.style={circle,draw=myblue,thick,inner sep=1pt,minimum size=12pt,font=\tiny},
  mx/.style={rectangle,draw=myblue,thick,inner sep=2pt,minimum size=11pt,font=\tiny},
  mn/.style={diamond,draw=myred,thick,inner sep=1pt,minimum size=11pt,font=\tiny},
  gpnode/.style={rectangle,rounded corners=2pt,draw=myblue,thick,inner sep=2pt,minimum size=10pt,font=\tiny},
  ed/.style={-{Latex[length=1mm]},thin}
}
